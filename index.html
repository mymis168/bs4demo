一、BeautifulSoup4 簡介

1.1 什麼是 BeautifulSoup？

BeautifulSoup 是一個 Python 套件，用於從 HTML 和 XML 文件中提取資料。它提供了簡單的函數和方法，讓新手也能輕鬆地搜尋、定位和修改樹狀結構中的節點。

1.2 核心特點

  • 易用性：提供簡潔的 Pythonic API
  • 容錯能力：自動處理格式不良的 HTML
  • 多解析器支援：支援 html.parser, lxml, html5lib 等多種解析器
  • 國際化：支援多種字元編碼自動偵測
  • 社區活躍：開源專案，豐富的生態系

1.3 BeautifulSoup 在爬蟲流程中的位置

# 整體流程
requests.get(url)  ──→  取得 HTML 原始碼
         │
         ▼
BeautifulSoup(html, parser)  ──→  建立解析樹
         │
         ▼
find() / find_all()  ──→  提取目標資料
         │
         ▼
儲存 / 分析 / 後續處理

1.4 與其他工具比較

工具 優點 缺點
BeautifulSoup 易學易用、容錯強 純 Python 實現，速度較慢
lxml (直接) 速度極快 API 較複雜
Scrapy 完整爬蟲框架 學習曲線陡峭
selenium 支援 JavaScript 渲染 資源消耗大

5.2 進階搜尋技巧

from bs4 import BeautifulSoup
import re

html = """


    
歡迎

    
內容1

    
內容2

    第1頁
    第2頁


"""
soup = BeautifulSoup(html, "html.parser")

# ===== 使用正則表達式 =====

# 搜尋 class 符合正則的標籤
pattern = re.compile(r"^content")
results = soup.find_all("p", class_=pattern)

# 搜尋 href 符合正則的連結
pattern = re.compile(r"/page/\d+")
links = soup.find_all("a", href=pattern)

# ===== 使用 lambda 函數 =====

# 搜尋所有有 id 屬性的標籤
results = soup.find_all(lambda tag: tag.has_attr("id"))

# 搜尋特定條件的標籤
results = soup.find_all(lambda tag:
    tag.name == "a" and tag.has_attr("href"))

# ===== 多個條件組合 =====

# 同時滿足多個條件
results = soup.find_all("p", class_="content")

# 搜尋 class 包含特定值的標籤
results = soup.find_all("p", class_="content")

5.3 限制返回數量

# find_all() 支援 limit 參數
results = soup.find_all("p", limit=2)
# 最多返回 2 個結果

# find() 等同於 find_all(limit=1)
first_result = soup.find("p")
# 等同於
first_result = soup.find_all("p", limit=1)[0]

5.4 搜尋參數完整對照表

參數 說明 範例
標籤名稱 第一個位置參數指定標籤名 find_all("div")
class_ 按 class 屬性搜尋 find_all("p", class_="text")
id 按 id 屬性搜尋 find_all(id="main")
text 按文字內容搜尋 find_all(text="目標")
attrs 自訂屬性字典 find_all(attrs={"data-type": "a"})
recursive 是否只搜尋子節點 find_all("p", recursive=False)
limit 限制返回數量 find_all("p", limit=5)
string 按文字搜尋（同 text） find_all(string="目標")

8.1 範例一：爬取新聞標題

import requests
from bs4 import BeautifulSoup
import json

def scrape_news(url):
    """從新聞網站抓取標題"""
    headers = {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
    }
    
    response = requests.get(url, headers=headers)
    response.raise_for_status()
    
    soup = BeautifulSoup(response.content, "html.parser")
    
    news_data = []
    
    # 假設新聞列表在 
 中
    news_items = soup.find_all("article", class_="news-item")
    
    for item in news_items:
        title_tag = item.find("h2")
        link_tag = item.find("a")
        date_tag = item.find("span", class_="date")
        
        if title_tag and link_tag:
            news_data.append({
                "title": title_tag.get_text(strip=True),
                "url": link_tag.get("href"),
                "date": date_tag.get_text(strip=True) if date_tag else None
            })
    
    return news_data

# 使用方式

8.7 範例七：動態網站資料提取

from bs4 import BeautifulSoup
import json

def extract_api_from_html(html_content):
    """從 HTML 中提取嵌入的 JSON API 資料"""
    soup = BeautifulSoup(html_content, "html.parser")
    
    # 許多現代網站將資料嵌入在 script 標籤中
    # 尋找包含 JSON 資料的 script 標籤
    script_tags = soup.find_all("script")
    
    for script in script_tags:
        if script.string and "window.__INITIAL_STATE__" in script.string:
            # 提取 JSON 資料
            json_str = script.string.split("window.__INITIAL_STATE__ = ")[1]
            json_str = json_str.rstrip(";")
            data = json.loads(json_str)
            return data
    
    return None

# 使用方式

#     data = extract_api_from_html(f.read())

8.8 範例八：完整爬蟲類別封裝

import requests
from bs4 import BeautifulSoup
import time
import json

class WebScraper:
    """
    通用網頁爬蟲類別
    
    使用範例:
        scraper = WebScraper(delay=1)
        scraper.add_request("https://example.com/news", "news_data.json")
        scraper.add_request("https://example.com/products", "products_data.json")
        scraper.execute()
    """
    
    def __init__(self, delay=1, timeout=10, user_agent=None):
        self.delay = delay
        self.timeout = timeout
        self.session = requests.Session()
        
        if user_agent:
            self.session.headers["User-Agent"] = user_agent
        else:
            self.session.headers["User-Agent"] = (
                "Mozilla/5.0 (Windows NT 10.0; Win64; x64)"
                " AppleWebKit/537.36 (KHTML, like Gecko)"
                " Chrome/120.0.0.0 Safari/537.36"
            )
    
    def fetch(self, url):
        """發送 GET 請求並返回 BeautifulSoup 物件"""
        try:
            response = self.session.get(url, timeout=self.timeout)
            response.raise_for_status()
            return BeautifulSoup(response.content, "html.parser")
        except requests.RequestException as e:
            print(f"請求失敗: {url} - {e}")
            return None
    
    def extract_items(self, soup, selector):
        """根據 CSS 選擇器提取項目列表"""
        elements = soup.select(selector)
        items = []
        
        for el in elements:
            item = {}
            # 根據實際網站結構提取欄位
            for field, field_selector in self.fields.items():
                tag = el.select_one(field_selector) if isinstance(field_selector, str) else el
                if tag:
                    item[field] = tag.get_text(strip=True)
                else:
                    item[field] = None
            items.append(item)
        
        return items
    
    def save_json(self, data, filename):
        """將資料儲存為 JSON 檔案"""
        with open(filename, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
        print(f"已儲存 {len(data)} 筆資料到 {filename}")
    
    def run(self, url, output_file, selector_config):
        """執行爬取流程"""
        self.fields = selector_config
        soup = self.fetch(url)
        
        if soup:
            items = self.extract_items(soup, "article")
            self.save_json(items, output_file)
        
        time.sleep(self.delay)

import requests
from bs4 import BeautifulSoup

def safe_scrape(url):
    """安全地爬取網頁"""
    try:
        response = requests.get(url, timeout=10)
        response.raise_for_status()
        
        soup = BeautifulSoup(response.content, "html.parser")
        
        # 檢查是否找到目標元素
        target = soup.find("div", class_="content")
        if not target:
            print("未找到目標內容")
            return None
        
        return target.get_text()
        
    except requests.Timeout:
        print("請求逾時")
    except requests.HTTPError as e:
        print(f"HTTP 錯誤: {e}")
    except Exception as e:
        print(f"發生錯誤: {e}")
    
    return None

8.2 範例二：下載圖片

import requests
from bs4 import BeautifulSoup
import os

def download_images(url, save_dir="images"):
    """從網頁下載所有圖片"""
    os.makedirs(save_dir, exist_ok=True)
    
    response = requests.get(url)
    soup = BeautifulSoup(response.content, "html.parser")
    
    images = soup.find_all("img")
    
    for img in images:
        src = img.get("src")
        if not src:
            continue
        
        # 處理相對路徑
        if src.startswith("//"):
            src = "https:" + src
        elif src.startswith("/"):
            from urllib.parse import urljoin
            src = urljoin(url, src)
        
        if src.startswith(("http://", "https://")):
            try:
                img_response = requests.get(src)
                # 從 URL 提取檔名
                filename = os.path.basename(src)
                save_path = os.path.join(save_dir, filename)
                
                with open(save_path, "wb") as f:
                    f.write(img_response.content)
                print(f"已下載: {filename}")
            except requests.RequestException as e:
                print(f"下載失敗: {src} - {e}")

import requests
from bs4 import BeautifulSoup

def scrape_login_required_site():
    """使用 session 保持登入狀態"""
    
    session = requests.Session()
    
    # 1. 先取得登入頁面（獲取 CSRF token）
    login_page = session.get("https://example.com/login")
    soup = BeautifulSoup(login_page.content, "html.parser")
    
    # 提取 CSRF token
    csrf_token = soup.find("input", name="_csrf")
    if csrf_token:
        csrf_value = csrf_token["value"]
    else:
        print("未找到 CSRF token")
        return
    
    # 2. 發送登入請求
    login_data = {
        "username": "your_username",
        "password": "your_password",
        "_csrf": csrf_value
    }
    
    login_response = session.post(
        "https://example.com/login",
        data=login_data,
        headers={"Referer": "https://example.com/login"}
    )
    
    # 3. 檢查是否登入成功
    if "歡迎" in login_response.text:
        print("登入成功！")
        
        # 4. 爬取登入後的頁面
        dashboard = session.get("https://example.com/dashboard")
        soup = BeautifulSoup(dashboard.content, "html.parser")
        
        # 解析 dashboard 內容
        print(soup.get_text())
    else:
        print("登入失敗")

import requests
from bs4 import BeautifulSoup
import csv

def table_to_csv(url, output_file="output.csv"):
    """將網頁表格轉換為 CSV 檔案"""
    
    response = requests.get(url)
    soup = BeautifulSoup(response.content, "html.parser")
    
    tables = soup.find_all("table")
    
    if not tables:
        print("未找到表格")
        return
    
    for table_idx, table in enumerate(tables):
        rows = table.find_all("tr")
        
        if not rows:
            continue
        
        with open(output_file, "w", newline="", encoding="utf-8") as f:
            writer = csv.writer(f)
            
            for row in rows:
                cells = row.find_all(["th", "td"])
                row_data = [cell.get_text(strip=True) for cell in cells]
                writer.writerow(row_data)
        
        print(f"已儲存: {output_file}")

8.6 範例六：分頁爬取

import requests
from bs4 import BeautifulSoup
import time

def scrape_pagination(base_url, total_pages=5):
    """爬取多頁資料"""
    
    all_data = []
    
    for page in range(1, total_pages + 1):
        url = f"{base_url}?page={page}"
        print(f"正在爬取第 {page} 頁...")
        
        try:
            response = requests.get(url)
            soup = BeautifulSoup(response.content, "html.parser")
            
            items = soup.find_all("div", class_="item")
            
            if not items:
                print(f"第 {page} 頁沒有資料，停止爬取")
                break
            
            for item in items:
                title = item.find("h2")
                if title:
                    all_data.append(title.get_text(strip=True))
            
            # 避免請求過快，暫停 1 秒
            time.sleep(1)
            
        except requests.RequestException as e:
            print(f"爬取第 {page} 頁失敗: {e}")
            break
    
    print(f"共爬取 {len(all_data)} 筆資料")
    return all_data
